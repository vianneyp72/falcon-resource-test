# Part 6: UI (Tasks 19-22)

The UI is React 19 + Vite in plain JS/JSX, matching `webapp/`, with uPlot for charts. Size matters because the bundle is embedded in the binary, which counts against the 25 MB image budget, and it's what the browser downloads over `kubectl port-forward`. So:
- Runtime dependencies are only `react`, `react-dom`, and `uplot`. There's no router library (a 30-line history hook covers four routes), no UI kit, and no date library.
- Fonts: system font stack only. No font files are bundled (the repo is public, and Sharp Sans is a licensed brand asset).
- Build output is pre-gzipped by an inline Vite plugin (Node `zlib`, no plugin dependency), and Task 15's static handler serves the `.gz` siblings.
- Target: **under 120 KB gzipped JS**, checked in Task 22.

All work in this part happens in `ui/`.

**JSON shapes the UI reads** (from Parts 2 and 4):
- `/api/live` → `Live` (Task 15).
- `/api/series` → `{step, components: {<id>: [{ts, cpu_m, mem_bytes}]}, cluster: [{ts, alloc_cpu_m, alloc_mem, used_cpu_m, used_mem}]}`.
- `Report`: `{from, to, headline, cluster, components, nodes, checks, config, self, method}`. Stats are `{avg, p50, p95, peak}`. `method.window` and `method.gaps` are Go durations in **nanoseconds**.
- `Bundle`: `{report, series}`, where `series` is `{<component>: [{ts, cpu_m, mem_bytes}]}`.
- `model.ContainerSpec` (`report.config[]`) has no JSON tags, so its keys are capitalized: `Component`, `Namespace`, `Pod`, `Container`, `Node`, `Image`, `Version`, `Backend`, `PriorityClass`, `ReqCPUm`, `ReqMem`, `LimCPUm`, `LimMem`.
- `Assessment`: `{id, kind, started_at, ends_at, status, created_at}`, with RFC 3339 times.
- A check is `{id, title, status, summary, details}`, where `status` is one of `pass`, `fail`, `review`, `info`, or `not_measured`.

```js
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'
import { gzipSync } from 'node:zlib'

// Writes a .gz sibling for every text asset. The Go server serves those
// as-is, so the binary never compresses per request.
function pregzip() {
  return {
    name: 'pregzip',
    apply: 'build',
    generateBundle(_, bundle) {
      for (const [name, file] of Object.entries(bundle)) {
        if (!/\.(js|css|html|svg)$/.test(name)) continue
        const src = file.type === 'chunk' ? file.code : file.source
        this.emitFile({ type: 'asset', fileName: name + '.gz', source: gzipSync(src, { level: 9 }) })
      }
    },
  }
}

export default defineConfig({
  plugins: [react(), pregzip()],
  build: {
    outDir: '../internal/web/dist',
    emptyOutDir: true,
    assetsDir: 'assets',
    sourcemap: false,
    target: 'es2022',
  },
  server: {
    // `npm run dev` against a port-forwarded tool:
    //   kubectl -n falcon-resource-test port-forward svc/falcon-resource-test 8080:80
    proxy: { '/api': 'http://localhost:8080' },
  },
  test: { environment: 'jsdom' },
})
```

- [ ] **Step 6: Implement `src/lib/format.js`**

```js
const MiB = 2 ** 20
const GiB = 2 ** 30

export function cpu(m) {
  if (m === 0) return '0m'
  if (m < 1) return '<1m'
  if (m < 1000) return `${Math.round(m)}m`
  return `${(m / 1000).toFixed(2)} cores`
}

export function mem(b) {
  if (b >= GiB) return `${(b / GiB).toFixed(2)} GiB`
  const mib = b / MiB
  return `${mib >= 100 || mib === 0 ? Math.round(mib) : mib.toFixed(1)} MiB`
}

export function pct(n) {
  if (n === 0) return '0%'
  if (n < 0.01) return '<0.01%'
  return `${n < 10 ? n.toFixed(2) : n.toFixed(1)}%`
}

// Go time.Duration marshals as integer nanoseconds.
export function dur(ns) {
  let s = Math.round(ns / 1e9)
  if (s < 60) return `${s}s`
  const h = Math.floor(s / 3600)
  const m = Math.round((s % 3600) / 60)
  if (h === 0) return `${m}m`
  return m === 0 ? `${h}h` : `${h}h ${m}m`
}

// Go's zero time.Time marshals as year 1, so anything before 2000 means "never".
export function ago(iso, now = new Date()) {
  const t = new Date(iso)
  if (t.getUTCFullYear() < 2000) return 'never'
  const s = Math.max(0, Math.round((now - t) / 1000))
  if (s < 60) return `${s}s ago`
  if (s < 3600) return `${Math.floor(s / 60)}m ago`
  if (s < 86400) return `${Math.floor(s / 3600)}h ago`
  return `${Math.floor(s / 86400)}d ago`
}

export function when(iso) {
  return new Date(iso).toLocaleString(undefined, { dateStyle: 'medium', timeStyle: 'short' })
}

export const STATUS_LABEL = {
  pass: 'Pass',
  fail: 'Fail',
  review: 'Review',
  info: 'Info',
  not_measured: 'Not measured',
}
```

```js
  const res = await fetch(path, opts)
  const body = await res.json().catch(() => ({}))
```

`src/lib/usePoll.js` polls only while the tab is visible. A forgotten background tab costs the cluster nothing:

```js
import { useCallback, useEffect, useRef, useState } from 'react'

export function usePoll(fn, ms, deps = []) {
  const [data, setData] = useState(null)
  const [error, setError] = useState(null)
  const fnRef = useRef(fn)
  fnRef.current = fn

  const load = useCallback(async () => {
    try {
      setData(await fnRef.current())
      setError(null)
    } catch (e) {
      setError(e.message)
    }
  }, [])

  useEffect(() => {
    let timer
    const tick = () => {
      clearTimeout(timer)
      if (document.visibilityState !== 'visible') return
      load().finally(() => {
        if (ms) timer = setTimeout(tick, ms)
      })
    }
    tick()
    document.addEventListener('visibilitychange', tick)
    return () => {
      clearTimeout(timer)
      document.removeEventListener('visibilitychange', tick)
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [load, ms, ...deps])

  return { data, error, reload: load }
}
```

`src/lib/route.jsx` is a minimal history router that covers the four routes:

```js
import { useEffect, useState } from 'react'

export function navigate(to) {
  if (to === location.pathname) return
  history.pushState(null, '', to)
  dispatchEvent(new PopStateEvent('popstate'))
  scrollTo(0, 0)
}

export function usePath() {
  const [path, setPath] = useState(location.pathname)
  useEffect(() => {
    const on = () => setPath(location.pathname)
    addEventListener('popstate', on)
    return () => removeEventListener('popstate', on)
  }, [])
  return path
}

// match('/assessments/:id', '/assessments/3') -> { id: '3' }
export function match(pattern, path) {
  const p = pattern.split('/')
  const a = path.split('/')
  if (p.length !== a.length) return null
  const params = {}
  for (let i = 0; i < p.length; i++) {
    if (p[i].startsWith(':')) params[p[i].slice(1)] = decodeURIComponent(a[i])
    else if (p[i] !== a[i]) return null
  }
  return params
}

export function Link({ to, children, ...rest }) {
  const onClick = (e) => {
    if (e.metaKey || e.ctrlKey || e.shiftKey || e.button !== 0) return
    e.preventDefault()
    navigate(to)
  }
  return <a href={to} onClick={onClick} {...rest}>{children}</a>
}
```

```jsx
import { useState } from 'react'
import { Link, match, usePath } from './lib/route'
import Live from './pages/Live'
import Assessments from './pages/Assessments'
import Report from './pages/Report'
import Setup from './pages/Setup'

const NAV = [
  ['/', 'Live'],
  ['/assessments', 'Assessments'],
  ['/setup', 'Setup'],
]

function ThemeToggle() {
  const [theme, setTheme] = useState(document.documentElement.dataset.theme)
  const flip = () => {
    const next = theme === 'dark' ? 'light' : 'dark'
    document.documentElement.dataset.theme = next
    localStorage.setItem('ff-theme', next)
    setTheme(next)
  }
  return (
    <button className="btn btn--ghost" onClick={flip} aria-label="Toggle theme">
      {theme === 'dark' ? 'Light' : 'Dark'}
    </button>
  )
}

function Page({ path }) {
  if (path === '/') return <Live />
  if (path === '/assessments') return <Assessments />
  if (path === '/setup') return <Setup />
  const m = match('/assessments/:id', path)
  if (m) return <Report id={m.id} />
  return <p className="muted">Page not found. <Link to="/">Back to Live</Link></p>
}

export default function App() {
  const path = usePath()
  const active = (to) => (to === '/' ? path === '/' : path.startsWith(to))
  return (
    <>
      <header className="topbar">
        <Link to="/" className="brand">
          <span className="brand__mark" aria-hidden="true" />
          Falcon <b>Resource Test</b>
        </Link>
        <nav className="nav">
          {NAV.map(([to, label]) => (
            <Link key={to} to={to} className={active(to) ? 'nav__item nav__item--active' : 'nav__item'}>
              {label}
            </Link>
          ))}
        </nav>
        <ThemeToggle />
      </header>
      <main className="main">
        <Page path={path} />
      </main>
    </>
  )
}
```

`/api/series` returns one point list per component, and the lists don't always share timestamps (a pod restarts, or a component is installed mid-window). uPlot needs a single x array, so `stack()` aligns them on the union of timestamps, fills missing points with 0, and builds cumulative layers for the stacked area.

```js
      { ts: 115, cpu_m: 6, mem_bytes: 60 },
    ],
    'daemonset-sensor': [
      { ts: 115, cpu_m: 20, mem_bytes: 200 },
      { ts: 130, cpu_m: 30, mem_bytes: 300 },
    ],
  },

describe('stack', () => {
  it('aligns on the union of timestamps in component order', () => {
    const s = stack(resp, 'cpu_m')
    expect(s.ts).toEqual([100, 115, 130])
    expect(s.layers.map((l) => l.component)).toEqual(['daemonset-sensor', 'kac'])
    expect(s.layers[0].raw).toEqual([0, 20, 30])
    expect(s.layers[1].raw).toEqual([5, 6, 0])
```

- [ ] **Step 2: Implement `src/lib/series.js`**

```js
// Display order and colors, matching model.FalconComponents. The DaemonSet
// Sensor sits at the bottom of the stack because it's usually the largest.
export const COMPONENTS = [
  { id: 'daemonset-sensor', label: 'DaemonSet Sensor', color: '#ED1C24' },
  { id: 'kac', label: 'KAC', color: '#5b9cf5' },
  { id: 'iar', label: 'IAR', color: '#3ecfba' },
  { id: 'injector', label: 'Container Sensor Injector', color: '#f0a830' },
  { id: 'operator', label: 'Falcon Operator', color: '#34d27b' },
]

export const meta = (id) => COMPONENTS.find((c) => c.id === id) || { id, label: id, color: '#8f8a9e' }

export function stack(resp, metric) {
  const present = COMPONENTS.filter((c) => resp.components[c.id]?.length)
  const set = new Set()
  for (const c of present) for (const p of resp.components[c.id]) set.add(p.ts)
  const ts = [...set].sort((a, b) => a - b)
  const at = new Map(ts.map((t, i) => [t, i]))

  const layers = []
  let below = new Array(ts.length).fill(0)
  for (const c of present) {
    const raw = new Array(ts.length).fill(0)
    for (const p of resp.components[c.id]) raw[at.get(p.ts)] = p[metric]
    const cum = raw.map((v, i) => v + below[i])
    layers.push({ component: c.id, raw, cum })
    below = cum
  }
  return { ts, layers }
}

// Cluster-wide usage (all workloads) as % of allocatable, on the stack's
// timestamps. It's drawn against a right-hand axis so a reader can see whether
// Falcon's usage follows cluster load.
export function load(cluster, metric, ts) {
  if (!cluster?.length) return null
  const [used, alloc] = metric === 'cpu_m' ? ['used_cpu_m', 'alloc_cpu_m'] : ['used_mem', 'alloc_mem']
  const by = new Map(cluster.map((p) => [p.ts, p[alloc] ? (p[used] / p[alloc]) * 100 : null]))
  return ts.map((t) => by.get(t) ?? null)
}
```

- [ ] **Step 3: Chart component**

`src/components/Chart.jsx` wraps uPlot. Stacking is drawn as cumulative lines, each filled to zero, rendered top layer first so each lower layer paints over the one above it. The legend and tooltip show each component's own value (`raw`), not the cumulative one. Two optional overlays:
- `reserved` adds a dashed horizontal line for the sum of Falcon's requests, i.e. what the scheduler has set aside for Falcon.
- `load` adds a thin gray line for cluster usage by all workloads, on a 0-100% right axis, so the reader can see whether Falcon tracks the workload.

```jsx
import { useEffect, useRef } from 'react'
import uPlot from 'uplot'
import { cpu, mem } from '../lib/format'
import { meta } from '../lib/series'

const css = (name) => getComputedStyle(document.documentElement).getPropertyValue(name).trim()

export default function Chart({ data, metric, reserved, load, height = 260 }) {
  const ref = useRef(null)
  const fmt = metric === 'cpu_m' ? cpu : mem

  useEffect(() => {
    const el = ref.current
    if (!el || !data.ts.length) return
    const layers = [...data.layers].reverse() // top of stack drawn first
    const axis = { stroke: css('--text-muted'), grid: { stroke: css('--border'), width: 1 }, ticks: { show: false } }
    const series = [
      {},
      ...layers.map((l) => {
        const m = meta(l.component)
        return {
          label: m.label,
          stroke: m.color,
          fill: m.color + '55',
          width: 1.5,
          points: { show: false },
          value: (u, v, si, i) => (i == null ? '--' : fmt(l.raw[i])),
        }
      }),
    ]
    const cols = [data.ts, ...layers.map((l) => l.cum)]
    if (reserved > 0) {
      series.push({ label: 'Reserved (requests)', stroke: css('--text-muted'), dash: [6, 4], width: 1, points: { show: false }, value: () => fmt(reserved) })
      cols.push(data.ts.map(() => reserved))
    }
    const axes = [axis, { ...axis, size: 70, values: (u, vals) => vals.map(fmt) }]
    if (load) {
      series.push({ label: 'Cluster load (all workloads)', scale: '%', stroke: css('--text-muted'), width: 1, spanGaps: true, points: { show: false }, value: (u, v) => (v == null ? '--' : `${v.toFixed(0)}%`) })
      cols.push(load)
      axes.push({ ...axis, scale: '%', side: 1, size: 50, grid: { show: false }, values: (u, vals) => vals.map((v) => `${v}%`) })
    }
    const opts = {
      width: el.clientWidth,
      height,
      series,
      axes,
      scales: { y: { range: (u, min, max) => [0, Math.max(max, reserved || 0) * 1.1 || 1] }, '%': { range: [0, 100] } },
      cursor: { points: { show: false } },
      legend: { live: true },
    }
    const plot = new uPlot(opts, cols, el)
    const ro = new ResizeObserver(() => plot.setSize({ width: el.clientWidth, height }))
    ro.observe(el)
    return () => {
      ro.disconnect()
      plot.destroy()
    }
  }, [data, metric, reserved, load, height, fmt])

  if (!data.ts.length) return <div className="empty">No samples yet for this range. The first appear within one collection interval.</div>
  return <div ref={ref} className="chart" />
}
```

The plot rebuilds whenever new data arrives (every 15s at most). At a few hundred points that takes well under a millisecond, and it's simpler than patching series in place.

```jsx
import { Link } from '../lib/route'

export function banners(live) {
  const out = []
  if (live.running) {
    out.push({ id: 'running', kind: 'info', text: 'Assessment running. Results freeze when it ends; you can close this tab.', link: `/assessments/${live.running.id}` })
  }
  if (!live.metrics_ok) {
    out.push({
      id: 'metrics',
      kind: 'danger',
      text: `Can't read usage from the Kubernetes Metrics API, so CPU and memory show 0. Install metrics-server (kubectl apply -f https://github.com/kubernetes-sigs/metrics-server/releases/latest/download/components.yaml), then wait one minute. Error: ${live.metrics_error || 'unknown'}`,
    })
  }
  if (!live.falcon_found) {
    out.push({
      id: 'no-falcon',
      kind: 'warning',
      text: 'No Falcon components found. The tool looks for the DaemonSet Sensor, KAC, IAR, injector, and operator by image name in all namespaces it can read. Check the Setup page for missing permissions.',
      link: '/setup',
      linkText: 'Open Setup',
    })
  }
  if (live.falcon_api && live.falcon_api !== 'off' && live.falcon_api !== 'ok') {
    out.push({ id: 'falcon-api', kind: 'warning', text: `Sensor health from the Falcon API is unavailable: ${live.falcon_api}` })
  }
  if (live.disk?.near) {
    out.push({ id: 'disk', kind: 'warning', text: 'Storage is over 80% full, so the oldest history is being dropped early. Raise persistence.size (or the emptyDir sizeLimit) to keep the full retention window.' })
  }
  return out
}

export default function Banners({ live }) {
  return banners(live).map((b) => (
    <div key={b.id} className={`banner banner--${b.kind}`} role={b.kind === 'danger' ? 'alert' : 'status'}>
      {b.text}{' '}
      {b.link && <Link to={b.link}>{b.linkText || 'View progress'}</Link>}
    </div>
  ))
}
```

- [ ] **Step 7: Implement `src/pages/Live.jsx`**

Poll rates: `/api/live` every 15s. The chart refreshes every 15s on the 15m and 1h ranges and every 60s on longer ones, because those use 5-minute buckets that don't change faster than that. The server caches `/api/live` for 10s, so several open tabs still cost one computation.

```jsx
import { useMemo, useState } from 'react'
import Banners from '../components/Banners'
import Chart from '../components/Chart'
import Checks from '../components/Checks'
import Kpi from '../components/Kpi'
import { api, nowSec } from '../lib/api'
import { ago, cpu, mem, pct } from '../lib/format'
import { usePoll } from '../lib/usePoll'
import { load, meta, stack } from '../lib/series'

const RANGES = [
  ['15m', 900],
  ['1h', 3600],
  ['6h', 21600],
  ['24h', 86400],
]
const COMPONENT_STATUS = { ok: 'pass', degraded: 'review', not_running: 'fail' }

function freshness(live) {
  const age = (Date.now() - new Date(live.last_tick)) / 1000
  if (new Date(live.last_tick).getUTCFullYear() < 2000) return ['review', 'Waiting for first collection']
  if (age > live.interval_seconds * 3) return ['fail', `Collection stalled, last ${ago(live.last_tick)}`]
  return ['pass', `Collecting every ${live.interval_seconds}s, last ${ago(live.last_tick)}`]
}

export default function Live() {
  const [range, setRange] = useState(3600)
  const [metric, setMetric] = useState('cpu_m')
  const [drill, setDrill] = useState({}) // {} | {component} | {node}
  const { data: live, error } = usePoll(api.live, 15000)
  const { data: series } = usePoll(() => api.series(nowSec() - range, nowSec(), drill), range <= 3600 ? 15000 : 60000, [range, drill.component, drill.node])
  const stacked = useMemo(() => (series ? stack(series, metric) : { ts: [], layers: [] }), [series, metric])
  const loadLine = useMemo(() => load(series?.cluster, metric, stacked.ts), [series, metric, stacked])

  if (error && !live) return <div className="banner banner--danger">Can't reach the Falcon Resource Test API: {error}</div>
  if (!live) return <div className="empty">Loading...</div>

  const [tone, label] = freshness(live)
  // Reserved line: all Falcon requests, one component's, or (per node) none,
  // since requests per node aren't in the live view.
  const shown = live.components.filter((c) => !drill.component || c.component === drill.component)
  const reserved = drill.node ? 0 : shown.reduce((sum, c) => sum + (metric === 'cpu_m' ? c.req_cpu_m : c.req_mem) * c.pods, 0)
  const title = drill.component ? `${meta(drill.component).label} usage` : drill.node ? `Falcon usage on ${drill.node}` : 'Falcon usage, stacked by component'
  const nodes = [...live.nodes].sort((a, b) => b.cpu_share - a.cpu_share)
  const k = live.kpi

  return (
    <>
      <section className="pagehead">
        <div>
          <h1>{live.cluster}</h1>
          <div className="muted">
            <span className={`dot status-${tone}`} /> {label}
          </div>
        </div>
        <div className="pagehead__right">
          <span className="muted" title="CPU and memory used by Falcon Resource Test itself">
            This tool: {cpu(live.self.cpu_m)} / {mem(live.self.mem_bytes)}
          </span>
          {/* Run assessment button: added in Task 21 */}
        </div>
      </section>

      <Banners live={live} />

      <section className="grid">
        <Kpi label="Falcon CPU" value={cpu(k.cpu_m)} sub={`${pct(k.cpu_pct)} of cluster allocatable`} />
        <Kpi label="Falcon memory" value={mem(k.mem_bytes)} sub={`${pct(k.mem_pct)} of cluster allocatable`} />
        <Kpi
          label="DaemonSet Sensor pods"
          value={`${k.ready}/${k.desired}`}
          sub="ready / desired"
          tone={k.desired === 0 ? undefined : k.ready === k.desired ? 'pass' : 'review'}
        />
        <Kpi label="Restarts (24h)" value={k.restarts_24h} sub="all Falcon containers" tone={k.restarts_24h > 0 ? 'review' : 'pass'} />
        <Kpi label="Nodes" value={k.nodes} sub="reporting metrics" />
      </section>

      <section className="card">
        <div className="toolbar">
          <h2>{title}</h2>
          {(drill.component || drill.node) && <button className="btn btn--ghost" onClick={() => setDrill({})}>Show whole cluster</button>}
          <div className="seg" role="group" aria-label="Metric">
            <button aria-pressed={metric === 'cpu_m'} onClick={() => setMetric('cpu_m')}>CPU</button>
            <button aria-pressed={metric === 'mem_bytes'} onClick={() => setMetric('mem_bytes')}>Memory</button>
          </div>
          <div className="seg" role="group" aria-label="Range">
            {RANGES.map(([l, sec]) => (
              <button key={l} aria-pressed={range === sec} onClick={() => setRange(sec)}>{l}</button>
            ))}
          </div>
        </div>
        <Chart data={stacked} metric={metric} reserved={reserved} load={loadLine} />
        {range > live.retention_seconds && <p className="muted">Only {Math.round(live.retention_seconds / 3600)}h of history is kept. Enable persistence for longer.</p>}
      </section>

      <section className="card">
        <h2>Components</h2>
        <table>
          <thead>
            <tr>
              <th>Component</th><th>Status</th><th>Version</th><th className="num">Pods</th>
              <th className="num">CPU</th><th className="num">Memory</th>
              <th className="num">Request / pod</th><th className="num">Limit / pod</th>
            </tr>
          </thead>
          <tbody>
            {live.components.map((c) => (
              <tr key={c.component} className="clickable" onClick={() => setDrill({ component: c.component })} title="Show this component's chart">
                <td><span className="swatch" style={{ background: meta(c.component).color }} />{c.label}</td>
                <td><span className={`badge status-${COMPONENT_STATUS[c.status]}`}>{c.status.replace('_', ' ')}</span></td>
                <td className="mono">{c.version || '--'}</td>
                <td className="num">{c.ready_pods}/{c.pods}</td>
                <td className="num">{cpu(c.cpu_m)}</td>
                <td className="num">{mem(c.mem_bytes)}</td>
                <td className="num">{cpu(c.req_cpu_m)} / {mem(c.req_mem)}</td>
                <td className="num">{c.lim_cpu_m ? cpu(c.lim_cpu_m) : 'none'} / {c.lim_mem ? mem(c.lim_mem) : 'none'}</td>
              </tr>
            ))}
          </tbody>
        </table>
        {!live.components.length && (
          <div className="empty">
            No Falcon components discovered in {live.discovery.namespaces?.length ? live.discovery.namespaces.join(', ') : 'any readable namespace'}.
            If Falcon runs under a custom image name or namespace, set <code>discovery.namespaces</code> in the Helm values.
          </div>
        )}
      </section>

      <section className="card">
        <h2>Nodes, highest Falcon share first</h2>
        <table>
          <thead>
            <tr><th>Node</th><th className="num">Falcon CPU</th><th className="num">% of node CPU</th><th className="num">Falcon memory</th><th className="num">% of node memory</th></tr>
          </thead>
          <tbody>

                <td className="mono">{n.node}</td>
                <td className="num">{cpu(n.cpu_m)}</td>

      </section>
    </>
  )
}
```

```jsx
import { useEffect, useRef, useState } from 'react'
import { api } from '../lib/api'
import { navigate } from '../lib/route'

const SCANS = [
  ['5m', 'Quick look'],
  ['15m', 'Recommended for a call'],
  ['1h', 'Covers a load test'],
  ['24h', 'A full business day, the strongest evidence'],
]
const LOOKBACKS = ['1h', '6h', '24h']

export default function RunDialog({ open, onClose, retentionSeconds }) {
  const ref = useRef(null)
  const [mode, setMode] = useState('scan')
  const [value, setValue] = useState('15m')
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState('')

  useEffect(() => {
    const d = ref.current
    if (open && !d.open) d.showModal()
    if (!open && d.open) d.close()
  }, [open])

  const lookbacks = LOOKBACKS.filter((l) => parseInt(l) * 3600 <= retentionSeconds)
  const pick = (m, v) => {
    setMode(m)
    setValue(v)
  }

  const submit = async (e) => {
    e.preventDefault()
    setBusy(true)
    setError('')
    try {
      const a = await api.start(mode === 'scan' ? { duration: value } : { lookback: value })
      onClose()
      navigate(`/assessments/${a.id}`)
    } catch (err) {
      setError(err.message)
    } finally {
      setBusy(false)
    }
  }

  return (
    <dialog ref={ref} className="dialog" onClose={onClose}>
      <form onSubmit={submit}>
        <h2>Run an assessment</h2>
        <fieldset>
          <legend>Watch from now</legend>
          {SCANS.map(([v, hint]) => (
            <label key={v} className="option">
              <input type="radio" name="pick" checked={mode === 'scan' && value === v} onChange={() => pick('scan', v)} />
              <b>{v}</b> <span className="muted">{hint}</span>
            </label>
          ))}
        </fieldset>
        <fieldset>
          <legend>Or report on data already collected</legend>
          {lookbacks.map((v) => (
            <label key={v} className="option">
              <input type="radio" name="pick" checked={mode === 'lookback' && value === v} onChange={() => pick('lookback', v)} />
              <b>Last {v}</b> <span className="muted">Ready instantly</span>
            </label>
          ))}
          <p className="muted">If the tool started less than that long ago, the report covers what it has and says so.</p>
        </fieldset>
        <p className="muted">For the most useful numbers, run your normal or peak workload during the window. Leaving the page doesn't stop a scan.</p>
        {error && <div className="banner banner--danger">{error}</div>}
        <div className="dialog__actions">
          <button type="button" className="btn btn--ghost" onClick={onClose}>Cancel</button>
          <button type="submit" className="btn btn--primary" disabled={busy}>{mode === 'scan' ? `Start ${value} scan` : `Build report`}</button>

      </form>
    </dialog>
  )
}
```

- [ ] **Step 5: Report page**

`src/pages/Report.jsx` covers the three states. While a scan runs, it shows progress plus the results so far (from `/api/report` over `[started_at, now)`), so the reader isn't staring at a spinner for 15 minutes. Once complete, it renders the frozen bundle.

```jsx
import { useEffect, useState } from 'react'
import ReportView from '../components/ReportView'
import { api, nowSec } from '../lib/api'
import { dur, when } from '../lib/format'
import { Link } from '../lib/route'
import { usePoll } from '../lib/usePoll'

function Progress({ a }) {
  const [now, setNow] = useState(Date.now())
  useEffect(() => {
    const t = setInterval(() => setNow(Date.now()), 1000)
    return () => clearInterval(t)
  }, [])
  const start = new Date(a.started_at)
  const end = new Date(a.ends_at)
  const done = Math.min(1, (now - start) / (end - start))
  return (
    <section className="card">
      <h2>Scan in progress, {dur(Math.max(0, end - now) * 1e6)} left</h2>
      <div className="progress"><div className="progress__bar" style={{ width: `${done * 100}%` }} /></div>
      <p className="muted">Ends {when(a.ends_at)}. Leaving this page doesn't stop the scan. Results so far are below and become final when it ends.</p>
    </section>
  )
}

export default function Report({ id }) {
  const { data: d, error } = usePoll(() => api.assessment(id), 15000, [id])
  const running = d?.assessment.status === 'running'
  const { data: partial } = usePoll(() => (running ? partialReport(d.assessment) : null), running ? 15000 : 0, [running])

  if (error) return <div className="banner banner--danger">{error} <Link to="/assessments">Back to assessments</Link></div>
  if (!d) return <div className="empty">Loading...</div>
  const a = d.assessment

  return (
    <>
      <section className="pagehead">
        <div>
          <h1>Assessment #{a.id}</h1>
          <div className="muted"><Link to="/assessments">Assessments</Link> / {a.kind === 'scan' ? 'Timed scan' : 'Report on history'}</div>
        </div>
        {a.status === 'complete' && <a className="btn btn--primary" href={api.exportUrl(a.id)}>Export HTML</a>}
      </section>
      {running && <Progress a={a} />}
      {a.status === 'interrupted' && (
        <div className="banner banner--warning">
          This scan was interrupted because the tool restarted before it ended, so its window has a gap and no frozen report. Run it again. If the pod was OOMKilled or evicted, check the Setup page and the tool's own usage on Live.
        </div>
      )}
      {d.bundle && <ReportView bundle={d.bundle} />}
      {running && partial && <ReportView bundle={partial} />}
    </>
  )
}

function partialReport(a) {
  const from = Math.floor(new Date(a.started_at) / 1000)
  const to = nowSec()
  if (to - from < 60) return null // too little data to summarize yet
  return api.report(from, to)
}
```

```jsx
          <Row ok={r.falcon_found} title="Falcon components discovered">
            {r.falcon_found
              ? 'At least one Falcon workload was found.'
              : 'No Falcon workloads found. Confirm Falcon is installed (kubectl get ds -A | grep falcon) and that every permission below is granted.'}
          </Row>
        </ul>
      </section>

      <section className="card">
        <h2>Permissions</h2>
        <table>
          <thead>
            <tr><th>Resource</th><th>Verb</th><th>Used for</th><th>Status</th></tr>
          </thead>
          <tbody>
            {r.permissions.map((p) => (
              <tr key={`${p.group}/${p.resource}/${p.verb}`}>
                <td className="mono">{p.group ? `${p.resource}.${p.group}` : p.resource}</td>
                <td className="mono">{p.verb}</td>
                <td>{p.used_for}</td>
                <td><span className={`badge status-${p.allowed ? 'pass' : 'fail'}`}>{p.allowed ? 'granted' : 'denied'}</span></td>
              </tr>
            ))}
          </tbody>
        </table>
      </section>
    </>
  )
}
```

`Check again` calls `/api/setup` directly, but the server caches the result for 30s (Task 15). That's deliberate: repeated clicks don't fire a burst of SelfSubjectAccessReviews at the API server.

- [ ] **Step 2: Bundle size budget**

`scripts/check-ui-size.sh` fails the build if the gzipped JS grows past budget. It runs in `make build` and in CI (Task 27):

```bash
#!/usr/bin/env bash
# Fails if the embedded UI's gzipped JavaScript exceeds the budget.
set -euo pipefail
BUDGET_KB=${BUDGET_KB:-120}
dist="$(dirname "$0")/../internal/web/dist/assets"
total=0
for f in "$dist"/*.js.gz; do
  total=$((total + $(wc -c < "$f")))
done
kb=$((total / 1024))
echo "UI JavaScript: ${kb} KB gzipped (budget ${BUDGET_KB} KB)"
if [ "$kb" -gt "$BUDGET_KB" ]; then
  echo "UI bundle over budget. Check for a new dependency with: npx vite-bundle-visualizer" >&2
  exit 1
fi
```
